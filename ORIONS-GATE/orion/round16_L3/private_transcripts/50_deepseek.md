class MagicLoopManager:

        await websocket.accept()

            "type": "connection_established",
            "session_id": session_id,
            "timestamp": datetime.now().isoformat(),
            "message": "Magic Loop Active. Interruptions enabled."
        })
    
    async def disconnect(self, session_id: str):
        if session_id in self.active_connections:
            del self.active_connections[session_id]
        if session_id in self.interruption_queues:
            del self.interruption_queues[session_id]

            while True:

                    # Handle brain dump
                    await self.handle_brain_dump(
                        session_id,
                        message["content"],
                        message["mode"]
                    )

                    # Generate universal seed
