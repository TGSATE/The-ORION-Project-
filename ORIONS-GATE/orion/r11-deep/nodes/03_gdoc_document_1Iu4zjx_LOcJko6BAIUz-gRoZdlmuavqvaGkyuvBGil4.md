static double clamp01(double x) {
    if (x < 0.0) return 0.0;
    if (x > 1.0) return 1.0;
    return x;
}
static double clampf(double x, double a, double b) {
    if (x < a) return a;
    if (x > b) return b;
    return x;
}


static double h1_deviation(void) {
    int i;
    double acc = 0.0;
    for (i = 0; i < N_AXES; i++) {
        double d = M.ax[i].x - SPEC[i].keel;
        acc += d * d;
    }
    return sqrt(acc / N_AXES);
}


static void psi_update(void) {
    M.psi = h1_deviation();
}


/* ---- Schauberger : qualité du mouvement, pas un décor ----
 * Implosion = concentration centripète (énergie utile, ordre).
 * Explosion = dispersion centrifuge (chaleur perdue, écart au keel).
 * Œuf : section qui accélère le filet vers l'axe (hyperboloïde doux).
 * 4°C : densité max de l'eau = sol thermique.
 * Vortex : couple les axes libres en spirale, rappelle les locked.
 */
static double schau_gain(void) {
    double t_err = fabs(M.s.temp_c - T_CONST) / 20.0;
    double egg   = clampf(M.s.egg, 0.45, 1.60);
    double impl  = clamp01(M.s.implosion);
    double vort  = clamp01(M.s.vortex);
    /* gain > 0 aide le retour ; gain < 0 disperse */
    return (impl - 0.5) * 2.0 * egg * (1.0 - t_err) * (0.4 + 0.6 * vort);
}


static void schau_step(void) {
    /* le winding avance avec le vortex ; l'œuf module le rayon */
    double wdot = M.s.vortex * (0.7 + 0.3 * M.s.implosion) * M.s.egg;
    M.s.wind += DT * wdot;
    M.s.radius = clampf(1.0 / (0.35 + M.s.egg * M.s.implosion), 0.25, 2.4);


    if (M.s.flow == FLOW_IMPLODE) {
        M.s.implosion = clamp01(M.s.implosion + DT * 0.35);
        M.s.vortex    = clamp01(M.s.vortex    + DT * 0.25);
        M.s.temp_c    = M.s.temp_c + DT * 0.4 * (T_CONST - M.s.temp_c);
    } else if (M.s.flow == FLOW_EXPLODE) {
        M.s.implosion = clamp01(M.s.implosion - DT * 0.40);
        M.s.vortex    = clamp01(M.s.vortex    + DT * 0.15);
        M.s.temp_c    = M.s.temp_c + DT * 0.8; /* échauffement explosif */
        if (M.s.temp_c > 40.0) M.s.temp_c = 40.0;
    } else {
        M.s.vortex = clamp01(M.s.vortex * (1.0 - 0.08 * DT));
        M.s.temp_c = M.s.temp_c + DT * 0.15 * (T_CONST - M.s.temp_c);
    }
}


/* Simulation d'axes : ressort + amortissement + couplage + Schauberger */
static void axes_step(void) {
    int i;
    double force[N_AXES];
    double g = schau_gain();


    for (i = 0; i < N_AXES; i++) {
        double x = M.ax[i].x;
        double pull = -SPEC[i].k * (x - SPEC[i].keel);
        double fric = -SPEC[i].damp * M.ax[i].v;
        /* implosion resserre vers le keel ; explosion pousse au hasard d'axe */
        double sch = g * (SPEC[i].keel - x) * (SPEC[i].locked ? 1.4 : 0.7);
        force[i] = pull + fric + sch;
    }
    for (i = 0; i < N_COUPLE; i++) {
        int a = COUPLE_A[i], b = COUPLE_B[i];
        double dx = M.ax[b].x - M.ax[a].x;
        /* vortex : le couplage tourne (phase = winding) */
        double phase = sin(M.s.wind * 1.7 + i);
        double c = K_COUPLE * (0.5 + 0.5 * M.s.vortex) * dx * (0.6 + 0.4 * phase);
        force[a] += c;
        force[b] -= c;
    }
    for (i = 0; i < N_AXES; i++) {
        M.ax[i].v += DT * force[i];
        M.ax[i].x += DT * M.ax[i].v;
        M.ax[i].x  = clamp01(M.ax[i].x);
        /* murs visqueux */
        if (M.ax[i].x <= 0.0 || M.ax[i].x >= 1.0) M.ax[i].v *= -0.2;
    }
    schau_step();
    psi_update();
    M.ticks++;
}


static void psi0_return(void) {

    for (i = 0; i < N_AXES; i++) {
        M.ax[i].x = SPEC[i].keel;
        M.ax[i].v = 0.0;

    M.s.wind      = 0.0;
    M.s.temp_c    = T_CONST;
    M.s.radius    = 0.8;
    M.s.flow      = FLOW_HOLD;
    psi_update();
}

    M.s.temp_c = 12.0 + (rand() % 20);

    psi_update();
}

static char *trim(char *s) {
    char *e;
    while (*s && isspace((unsigned char)*s)) s++;
    if (!*s) return s;
    e = s + strlen(s) - 1;
    while (e > s && isspace((unsigned char)*e)) *e-- = 0;
    return s;
}


static int cmd_run(char *line) {
    char buf[256], *tok, *arg;
    int n;
    strncpy(buf, line, sizeof(buf) - 1);
    buf[sizeof(buf) - 1] = 0;
    tok = strtok(buf, " \t");
    if (!tok) return 1;
    arg = strtok(NULL, " \t");


    if (!strcmp(tok, "help") || !strcmp(tok, "?")) help();
    else if (!strcmp(tok, "state")) print_state();
    else if (!strcmp(tok, "axes")) print_axes();
    else if (!strcmp(tok, "schau") || !strcmp(tok, "schauberger")) print_schau();
    else if (!strcmp(tok, "step")) {
        n = arg ? atoi(arg) : 1;
        if (n < 1) n = 1;
        if (n > 5000) n = 5000;
        while (n--) axes_step();
        print_state();
    } else if (!strcmp(tok, "sim")) {
        n = arg ? atoi(arg) : 20;
        if (n < 1) n = 1;
        if (n > 2000) n = 2000;
        {
            int i;
            for (i = 1; i <= n; i++) {
                axes_step();
                if (i == n || i % 5 == 0)
                    printf("t=%lu psi=%.4f impl=%.2f vort=%.2f temp=%.1f\n",
                           M.ticks, M.psi, M.s.implosion, M.s.vortex, M.s.temp_c);
            }
        }
        print_axes();
    } else if (!strcmp(tok, "shove")) {
        shove((unsigned)time(NULL) ^ (unsigned)M.ticks);
        puts("shove + EXPLODE");
        print_state();
    } else if (!strcmp(tok, "psi0") || !strcmp(tok, "home")) {
        psi0_return();
        puts("returned to psi0 / 4C egg");
        print_state();
    } else if (!strcmp(tok, "implode")) {
        M.s.flow = FLOW_IMPLODE;
        puts("flow IMPLODE (centripète)");
        print_schau();
    } else if (!strcmp(tok, "explode")) {
        M.s.flow = FLOW_EXPLODE;
        puts("flow EXPLODE (centrifuge)");
        print_schau();
    } else if (!strcmp(tok, "hold")) {
        M.s.flow = FLOW_HOLD;
        puts("flow HOLD");
    } else if (!strcmp(tok, "vortex")) {
        if (arg) M.s.vortex = clamp01(atof(arg));
        print_schau();
    } else if (!strcmp(tok, "egg")) {
        if (arg) M.s.egg = clampf(atof(arg), 0.45, 1.60);
        print_schau();
    } else if (!strcmp(tok, "temp")) {
        if (arg) M.s.temp_c = clampf(atof(arg), 0.0, 40.0);
        print_schau();
    } else if (!strcmp(tok, "room")) {
        if (arg) M.room = abs(atoi(arg)) % N_ROOMS;
        printf("room %d %s\n", M.room, ROOMS[M.room]);
    } else if (!strcmp(tok, "allow")) {
        if (!arg) puts("usage: allow <acte>");
        else printf("%s -> %s\n", arg,
                    governor_allow(arg) ? "OPEN" : "REFUSED");
    } else if (!strcmp(tok, "atlas")) atlas_exam();
    else if (!strcmp(tok, "catalog")) print_catalog();
    else if (!strcmp(tok, "memo")) memo();
    else if (!strcmp(tok, "quit") || !strcmp(tok, "exit")) return 0;
    else printf("inconnu: %s  (help)\n", tok);
    return 1;
}

    double radius;       /* rayon du cœur */
    double circ;         /* circulation Γ */
    int    hand;         /* +1 droit, -1 gauche */
    double twist;        /* Tw  (tours) */
    double writhe;       /* Wr  */
    double link;         /* Lk ≈ Tw + Wr */
    double torsion;      /* τ moyenne ds */
} Filament;


typedef struct {
    Water    w;
    Filament a;          /* cyclone */
    Filament b;          /* anticyclone */
    double   egg;        /* 0.7..1.4 section Schauberger */
    double   implosion;  /* 0..1 */
    double   energy;     /* énergie utile relative */
    unsigned ticks;
} DualVortex;


static DualVortex V;


static double clampf(double x, double lo, double hi) {
    if (x < lo) return lo;
    if (x > hi) return hi;
    return x;
}


/* Densité de l'eau liquide, forme de Thiesen autour de 4°C (assez juste 0–40°C) */
static double water_rho(double T) {
    double g = 1.0 - 1.0e-5 * fabs(T - T_RHOMAX) * (T + 8.0);
    return 0.999975 * g;
}


/* Viscosité relative : plus froide = plus visqueuse, mais 4°C n'est pas un min */
static double water_mu(double T) {
    return exp(1800.0 * (1.0 / (T + 273.15) - 1.0 / 277.15));
}


/* Ordre des réseaux H-bond : max près de 4°C, détruit par la chaleur */
static double water_cluster(double T) {
    double d = (T - T_RHOMAX) / 12.0;
    return clampf(exp(-d * d), 0.05, 1.0);
}


static void water_update(Water *w, double T) {
    w->T = clampf(T, 0.0, 40.0);
    w->rho = water_rho(w->T);
    w->mu = water_mu(w->T);
    w->angle = H2O_DEG + 0.04 * (w->T - T_RHOMAX); /* faible ouverture thermique */
    w->offset = w->angle - TET_DEG;
    w->cluster = water_cluster(w->T);
}


static double norm3(double x, double y, double z) {
    return sqrt(x * x + y * y + z * z);
}


static void filament_helix(Filament *f, int hand, double R, double pitch, double phase) {
    int i;
    f->hand = hand;
    f->radius = 0.12;
    f->circ = 1.0 * hand;
    for (i = 0; i < N_SEG; i++) {
        double t = (double)i / N_SEG * 2.0 * PI;
        double z = pitch * (t / (2.0 * PI));
        f->s[i].x = R * cos(hand * t + phase);
        f->s[i].y = R * sin(hand * t + phase);
        f->s[i].z = z;
    }
    for (i = 0; i < N_SEG; i++) {
        int j = (i + 1) % N_SEG;
        double dx = f->s[j].x - f->s[i].x;
        double dy = f->s[j].y - f->s[i].y;
        double dz = f->s[j].z - f->s[i].z;
        /* fermeture torique : on ignore le saut de z pour la tangente locale */
        if (i == N_SEG - 1) dz = f->s[0].z + pitch - f->s[i].z;
        double n = norm3(dx, dy, dz);
        if (n < 1e-12) n = 1e-12;
        f->s[i].tx = dx / n; f->s[i].ty = dy / n; f->s[i].tz = dz / n;
    }
}


/* Torsion cinématique τ = (t × t') · t'' / |t'|²  — ici différence discrète */
static double filament_torsion(Filament *f) {
    int i;
    double acc = 0.0;
    for (i = 0; i < N_SEG; i++) {
        int j = (i + 1) % N_SEG, k = (i + 2) % N_SEG;
        double t1x = f->s[i].tx, t1y = f->s[i].ty, t1z = f->s[i].tz;
        double t2x = f->s[j].tx, t2y = f->s[j].ty, t2z = f->s[j].tz;
        double t3x = f->s[k].tx, t3y = f->s[k].ty, t3z = f->s[k].tz;
        double d1x = t2x - t1x, d1y = t2y - t1y, d1z = t2z - t1z;
        double d2x = t3x - t2x, d2y = t3y - t2y, d2z = t3z - t2z;
        /* (t1 × d1) · d2 */
        double cx = t1y * d1z - t1z * d1y;
        double cy = t1z * d1x - t1x * d1z;
        double cz = t1x * d1y - t1y * d1x;
        double trip = cx * d2x + cy * d2y + cz * d2z;
        double d1n = norm3(d1x, d1y, d1z);
        if (d1n > 1e-9) acc += trip / (d1n * d1n);
    }
    f->torsion = acc / N_SEG;
    return f->torsion;
}


/* Twist = (hand * torsion intégrée) en tours approximés */
static double filament_twist(Filament *f) {
    filament_torsion(f);
    f->twist = f->hand * f->torsion * N_SEG / (2.0 * PI);
    return f->twist;
}


/* Writhe discret (Gauss) entre le filament et lui-même, O(N²) */
static double filament_writhe(Filament *f) {
    int i, j;
    double wr = 0.0;
    for (i = 0; i < N_SEG; i++) {
        int i2 = (i + 1) % N_SEG;
        double r1x = f->s[i].x, r1y = f->s[i].y, r1z = f->s[i].z;
        double d1x = f->s[i2].x - r1x, d1y = f->s[i2].y - r1y, d1z = f->s[i2].z - r1z;
        for (j = i + 2; j < N_SEG; j++) {
            int j2 = (j + 1) % N_SEG;
            if (i == 0 && j == N_SEG - 1) continue;
            double r2x = f->s[j].x, r2y = f->s[j].y, r2z = f->s[j].z;
            double d2x = f->s[j2].x - r2x, d2y = f->s[j2].y - r2y, d2z = f->s[j2].z - r2z;
            double rx = r1x - r2x, ry = r1y - r2y, rz = r1z - r2z;
            double cx = d1y * d2z - d1z * d2y;
            double cy = d1z * d2x - d1x * d2z;
            double cz = d1x * d2y - d1y * d2x;
            double r = norm3(rx, ry, rz);
            if (r < 1e-8) continue;
            wr += (cx * rx + cy * ry + cz * rz) / (r * r * r);
        }
    }
    f->writhe = wr / (2.0 * PI);
    return f->writhe;
}


/* Linking number de Gauss entre A et B */
static double linking(const Filament *a, const Filament *b) {
    int i, j;
    double lk = 0.0;
    for (i = 0; i < N_SEG; i++) {
        int i2 = (i + 1) % N_SEG;
        double r1x = a->s[i].x, r1y = a->s[i].y, r1z = a->s[i].z;
        double d1x = a->s[i2].x - r1x, d1y = a->s[i2].y - r1y, d1z = a->s[i2].z - r1z;
        for (j = 0; j < N_SEG; j++) {
            int j2 = (j + 1) % N_SEG;
            double r2x = b->s[j].x, r2y = b->s[j].y, r2z = b->s[j].z;
            double d2x = b->s[j2].x - r2x, d2y = b->s[j2].y - r2y, d2z = b->s[j2].z - r2z;
            double rx = r1x - r2x, ry = r1y - r2y, rz = r1z - r2z;
            double cx = d1y * d2z - d1z * d2y;
            double cy = d1z * d2x - d1x * d2z;
            double cz = d1x * d2y - d1y * d2x;
            double r = norm3(rx, ry, rz);
            if (r < 1e-8) continue;
            lk += (cx * rx + cy * ry + cz * rz) / (r * r * r);
        }
    }
    return lk / (4.0 * PI);
}


static void vortex_init(double T) {
    memset(&V, 0, sizeof(V));
    water_update(&V.w, T);
    V.egg = 1.00;
    V.implosion = 0.70;
    filament_helix(&V.a, +1, 0.55, 1.6, 0.0);
    filament_helix(&V.b, -1, 0.55, 1.6, PI);
}


/* Conversion twist → writhe : le fluide « dépense » de la torsion en vrille spatiale */
static void convert_twist_writhe(Filament *f, double rate) {
    double give = rate * f->twist;
    f->twist  -= give;
    f->writhe += give;
    f->link = f->twist + f->writhe;
}


static void vortex_step(void) {
    double cold = V.w.cluster;
    double egg  = clampf(V.egg, 0.6, 1.4);
    double impl = clampf(V.implosion, 0.0, 1.0);
    /* plus proche de 4°C et plus œuf, plus la torsion se range (implose) */
    double order = cold * impl * egg;
    double heat  = (1.0 - cold);
    double Ra, pitch, phase;


    V.ticks++;
    filament_twist(&V.a);
    filament_twist(&V.b);
    filament_writhe(&V.a);
    filament_writhe(&V.b);


    /* Călugăreanu : Lk conservé si pas de coupure ; ici le dual couple les deux fils */
    convert_twist_writhe(&V.a, DT * (0.15 + 0.50 * order));
    convert_twist_writhe(&V.b, DT * (0.15 + 0.50 * order));


    /* la chaleur casse les clusters → rayon qui s'ouvre (explosion) */
    Ra    = (0.35 + 0.40 * (1.0 - impl) + 0.25 * heat) / egg;
    pitch = 1.2 + 0.8 * (1.0 - order);
    phase = 0.12 * V.ticks * DT * (0.5 + order);


    filament_helix(&V.a, +1, Ra, pitch, phase);
    filament_helix(&V.b, -1, Ra, pitch, phase + PI);


    /* énergie utile : densité × ordre × implosion / viscosité */
    V.energy = V.w.rho * order * (0.4 + 0.6 * impl) / (0.5 + 0.5 * V.w.mu);
    V.a.link = V.a.twist + V.a.writhe;
    V.b.link = V.b.twist + V.b.writhe;
}


static void print_water(void) {
    printf("EAU   T=%.2f C  rho=%.6f  mu_rel=%.3f  cluster=%.3f\n",
           V.w.T, V.w.rho, V.w.mu, V.w.cluster);
    printf("      H-O-H=%.2f deg   tetra=%.2f deg   offset=%+.2f deg\n",
           V.w.angle, TET_DEG, V.w.offset);
}


static void print_fil(const char *n, const Filament *f) {
    printf("%s  hand=%+d  Tw=%+.4f  Wr=%+.4f  Tw+Wr=%+.4f  tau=%+.4f\n",
           n, f->hand, f->twist, f->writhe, f->link, f->torsion);
}


static void print_all(void) {
    double lk = linking(&V.a, &V.b);
    printf("t=%u  egg=%.2f  impl=%.2f  energy=%.4f  Lk(A,B)=%+.4f\n",
           V.ticks, V.egg, V.implosion, V.energy, lk);
    print_water();
    print_fil("FIL+", &V.a);
    print_fil("FIL-", &V.b);
}

    }
}
